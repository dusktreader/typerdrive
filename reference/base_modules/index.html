typerdrive.config

Provide controls for the configuration of typerdrive itself.

Note

This is distinct from the functionality provided in the settings module.

The settings feature is for configuring the app built with typerdrive. Usually, settings reflect behavior specific to the app.

The config feature is for configuring typerdrive behavior and app metadata that typerdrive uses, such as the app name, version, and description. It should not be used to manage the app's own behavior.

class TyperdriveConfig(BaseModel):
    """
    Define the configurable attributes of `typerdrive`.

    This class uses the XDG Base Directory specification for storing application data:
    - Logs and settings use XDG_STATE_HOME (defaults to ~/.local/state)
    - Cache data uses XDG_CACHE_HOME (defaults to ~/.cache)
    """

    app_name: str = sys.argv[0].split("/")[-1]
    """
    The name of the app built on `typerdrive`. Use this setting if you want to override the default behavior of using
    `sys.argv[0]`.
    """

    version: str | None = None
    """
    The version of the app built on `typerdrive`.
    """

    description: str | None = None
    """
    A short description of the app built on `typerdrive`.
    """

    log_file_rotation: FileRotationSpec = "1 week"
    """
    Define the how often logs should be rotated. See the [`loguru` docs](
        https://loguru.readthedocs.io/en/stable/overview.html#easier-file-logging-with-rotation-retention-compression
    ) for more information.
    """

    log_file_retention: FileRetentionSpec = "1 month"
    """
    Define the how soon logs should be deleted. See the [`loguru` docs](
        https://loguru.readthedocs.io/en/stable/overview.html#easier-file-logging-with-rotation-retention-compression
    ) for more information.
    """

    log_file_compression: FileCompressionSpec = "tar.gz"
    """
    Define whether rotated logs should be compressed. See the [`loguru` docs](
        https://loguru.readthedocs.io/en/stable/overview.html#easier-file-logging-with-rotation-retention-compression
    ) for more information.
    """

    log_file_name: str = "app.log"
    """
    The name of the file where logs will be written. This file will be created in the `log_dir` directory.
    """

    console_width: int | None = None
    """
    Set the width of the console for `rich.console`. Will override automatic resizing behavior.
    """

    console_ascii_only: bool = False
    """
    If set, only use ascii characters in the console.
    """

    state_home: Path = Field(default_factory=xdg_state_home)
    """
    Base directory for state data, defaulting to XDG_STATE_HOME.
    """

    cache_home: Path = Field(default_factory=xdg_cache_home)
    """
    Base directory for cache data, defaulting to XDG_CACHE_HOME.
    """

    encryption_cache_ttl: int = Field(default=300, ge=0)
    """Seconds to cache the unwrapped identity on disk; zero disables caching."""

    @computed_field
    @property
    def app_state_home(self) -> Path:
        """
        Retrieve the app's state directory.
        """
        return self.state_home / self.app_name

    @computed_field
    @property
    def log_dir(self) -> Path:
        """
        Retrieve the directory where logs will be stored.
        """
        return self.app_state_home / "logs"

    @computed_field
    @property
    def settings_path(self) -> Path:
        """
        Retrieve the file where settings will be stored.
        """
        return self.app_state_home / "settings.json"

    @computed_field
    @property
    def app_cache_home(self) -> Path:
        """
        Retrieve the directory where the cache data will be stored.
        """
        return self.cache_home / self.app_name

    @computed_field
    @property
    def managed_cache_dir(self) -> Path:
        """
        Retrieve the directory where the `diskcache` data will be stored.

        This is kept separate from the rest of the app's cache directory so that other internal files can live
        alongside it without showing up in (or being removed by) the cache commands.
        """
        return self.app_cache_home / "managed"

    @computed_field
    @property
    def files_dir(self) -> Path:
        """
        Retrieve the directory where files will be stored.
        """
        return self.app_state_home / "files"

    @computed_field
    @property
    def database_path(self) -> Path:
        """
        Retrieve the file where the application database will be stored.
        """
        return self.app_state_home / "db.sqlite"

def set_typerdrive_config(**kwargs: Any):
    """
    Update the `TyperdriveConfig` with the provided key/values.
    """
    global _typerdrive_config
    combined_config = {**_typerdrive_config.model_dump(), **kwargs}

def to_context(ctx: typer.Context, name: str, val: TyperdriveManager) -> None:
    """
    Attach a `TyperdriveManager` object to the `TyperdriveContext` at the given `name`.
    """
    user_context = get_user_context(ctx)
    field_type = TyperdriveContext.__dataclass_fields__[name].type

    if get_origin(field_type) is UnionType:
        defined_types = [t for t in get_args(field_type) if t is not NoneType]
        require_condition(
            len(defined_types) == 1,
            "PANIC! TyperdriveContext fields must only have one type or None.",
            raise_exc_class=RuntimeError,
        )
        field_type = defined_types[0]

    # TODO: Get the type hinting on the next line right.
    ContextError.ensure_type(val, field_type, "Value is not of type any of the union types")  # type: ignore[arg-type]  # ty: ignore[invalid-argument-type]

    setattr(user_context, name, val)

def clear_directory(path: Path) -> int:
    """
    Delete all files and directories in the directory at the given path.

    If the target path does not exist or is not a directory, an exception will be raised. A symlink supplied as the root
    path is followed to clear its target directory, and the root symlink is preserved. Symlinks found inside the target
    directory are unlinked without following them or modifying their targets.
    """
    TyperdriveError.require_condition(path.exists(), f"Target {path=} does not exist")
    TyperdriveError.require_condition(path.is_dir(), f"Target {path=} is not a directory")
    with TyperdriveError.handle_errors(f"Failed to clear directory at {path=}"):
        return _clear_dir(path)

def is_child(path: Path, parent: Path) -> bool:
    """
    Return true if the given path is a child of the given parent.

    Both paths are resolved to handle symlinks before comparison.
    """
    # Resolve both paths to handle symlinks (e.g., /var -> /private/var on macOS)
    resolved_path = path.resolve()
    resolved_parent = parent.resolve()

    root_path = Path(resolved_path.parts[0])
    temp_path = resolved_path
    while temp_path != root_path:
        if temp_path == resolved_parent:
            return True
        temp_path = temp_path.parent
    return False

def render_directory(path: Path, is_root: bool = True) -> DirInfo:
    """
    Render a visualization of the directory at the given path using `rich.tree`.

    Omit symlink entries and never traverse their targets. Call this recursive function with `is_root=True`.
    """
    root_label: str
    if is_root:
        root_label = str(path)
        color = "[bold yellow]"
    else:
        root_label = escape(path.name)
        color = "[bold blue]"

    dir_info = DirInfo(
        tree=Tree(f"{color}📂 {root_label}"),
        file_count=0,
        total_size=0,
    )

    child_paths = sorted(
        (child_path for child_path in path.iterdir() if not child_path.is_symlink()),
        key=lambda p: (
            p.is_file(),
            p.name.lower(),
        ),
    )
    for child_path in child_paths:
        if child_path.is_dir():
            child_info = render_directory(child_path, is_root=False)
            dir_info.tree
            dir_info.tree.children.append(child_info.tree)
            dir_info.file_count += child_info.file_count
            dir_info.total_size += child_info.total_size
        else:
            file_size = child_path.stat().st_size
            icon = Text("📄 ")
            label = Text(escape(child_path.name), "green")
            info = Text(f" ({humanize.naturalsize(file_size)})", "blue")
            dir_info.tree.add(icon + label + info)
            dir_info.file_count += 1
            dir_info.total_size += file_size
    return dir_info

def tweak_env(**kwargs: str):
    """
    Temporarily alter environment variables for the lifetime of the context manager.

    All items in `kwargs` will be set in the environment.
    If there was exiting value in the given key, it will be restored after the context manager exits.
    If there was no existing value in the given key, it will have no value after the context manager exits.
    """
    old_vals: dict[str, str] = {}
    del_vals: list[str] = []
    for key, val in kwargs.items():
        old_val = os.environ.get(key, None)
        if old_val:
            old_vals[key] = old_val
        else:
            del_vals.append(key)
        os.environ[key] = val

    yield

    for key, val in old_vals.items():
        os.environ[key] = val
    for key in del_vals:
        del os.environ[key]

def pretty_annotation(annotation: Any) -> str:
    """Format built-in collection annotations and their type arguments for display."""
    origin = get_origin(annotation)
    if origin in (list, set, frozenset, tuple, dict):
        args = get_args(annotation)
        arg_names = ["..." if arg is Ellipsis else pretty_annotation(arg) for arg in args]
        return f"{origin.__name__}[{', '.join(arg_names)}]"
    if isinstance(annotation, type):
        return annotation.__name__
    return str(annotation)

        do_else:          A function that will be called if no exceptions were handled.
        do_finally:       A function that will always be called, even if an exception was handled.
        unwrap_message:   If true, "unwrap" the message so that newline characters are removed.
        debug:            If true, the message shown in the `terminal_message` will include the string representation of
                          the handled error. Not as suitable for end-users, and should only be used for debugging.
    """

    class _DefaultIgnoreException(Exception):
        """
        Define a special exception class to use for the default ignore behavior.

        Basically, this exception type can't be extracted from this method (easily), and thus could never actually
        be raised in any other context. This is only created here to preserve the `try/except/except/else/finally`
        structure.
        """

    ignore_exc_class = _DefaultIgnoreException if ignore_exc_class is None else ignore_exc_class

    def _decorate(func: WrappedFunction[P, T]) -> WrappedFunction[P, T]:
        rewriter = SignatureRewriter(func)

        def handle_error(err: Exception) -> NoReturn:
            try:
                final_message = reformat_exception(base_message, err)
            except Exception as msg_err:
                raise RuntimeError(f"Failed while formatting message: {repr(msg_err)}")

            trace = get_traceback()
            if do_except:
                do_except(
                    DoExceptParams(
                        err=err,
                        base_message=base_message,
                        final_message=final_message,
                        trace=trace,
                    )
                )

            subject: str | None = base_message
            footer: str | None = None
            if isinstance(err, TyperdriveError):
                if err.subject:
                    subject = err.subject
                if err.footer:
                    footer = err.footer
                message = err.message if debug else err.base_message or err.message
                if unwrap_message:
                    message = snick.unwrap(message)
                exit_code = err.exit_code
            else:
                message = str(err)
                exit_code = ExitCode.GENERAL_ERROR

            terminal_message(message, subject=f"[red]{subject}[/red]", footer=footer, error=True)
            raise typer.Exit(code=exit_code)

        async def await_result(awaitable: Awaitable[T]) -> T:
            try:
                result = await awaitable
            except ignore_exc_class:
                raise
            except handle_exc_class as err:
                handle_error(cast(Exception, err))
            else:
                if do_else:
                    do_else()
                return result
            finally:
                if do_finally:
                    do_finally()

        if iscoroutinefunction(func):

            async def wrapper(*args: P.args, **kwargs: P.kwargs) -> T:
                return await await_result(cast(Awaitable[T], func(*args, **kwargs)))
        else:

            def wrapper(*args: P.args, **kwargs: P.kwargs) -> T:
                return_value: T | Awaitable[T] | None = None
                try:
                    return_value = func(*args, **kwargs)
                    if isawaitable(return_value):
                        return cast(T, await_result(cast(Awaitable[T], return_value)))
                except ignore_exc_class:
                    raise
                except handle_exc_class as err:
                    handle_error(cast(Exception, err))
                else:
                    if do_else:
                        do_else()
                    return cast(T, return_value)
                finally:
                    if not isawaitable(return_value) and do_finally:
                        do_finally()

        rewriter.apply(wrapper)
        return cast(WrappedFunction[P, T], wrapper)

    return _decorate

def show_info(
    full: Annotated[bool, typer.Option("--full", help="Show the full Typerdrive configuration.")] = False,
):
    """Show app metadata and, optionally, the full Typerdrive configuration."""
    config = get_typerdrive_config()

    if full:
        config_info = Table(box=box.SIMPLE_HEAD)
        config_info.add_column("Setting", style="bold", no_wrap=True)
        config_info.add_column("Value")
        for key, value in config.model_dump().items():
            config_info.add_row(key, str(value))
        terminal_message(config_info, subject="Typerdrive Config")
        return

    version = config.version or "unknown"
    info = Table.grid(padding=(0, 2))
    info.add_column(style="bold", no_wrap=True)
    info.add_column()
    info.add_row("Name", config.app_name)
    info.add_row("Version", version)
    if config.description:
        info.add_row("Description", config.description)
    terminal_message(info, subject="App Information")

def show_version(
    terse: Annotated[bool, typer.Option("--terse", help="Print only the unformatted version string.")] = False,
):
    """Show the app version."""
    config = get_typerdrive_config()

    if terse:
        message = config.version or "unknown"
    elif not config.version:
        message = f"{config.app_name} (version unknown)"
    else:
        message = f"{config.app_name} v{config.version}"

    typer.echo(message)

def get_version() -> str:
    """
    Try to get version from metadata, then fallback to `pyproject.toml`, then fallback to "unknown".
    """
    try:
        return get_version_from_metadata()
    except metadata.PackageNotFoundError:
        try:
            return get_version_from_pyproject()
        except (FileNotFoundError, KeyError):
            return "unknown"
